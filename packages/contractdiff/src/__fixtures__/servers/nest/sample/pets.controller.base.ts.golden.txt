import { Body, Get, Param, Post, Query, Res } from '@nestjs/common';

import { coerce, invalid, problem, type Reply, send } from './contract-runtime.js';
import { type paths } from "./types.js";
import { operations as validators } from "./validators.js";

type CreatePetOperation = paths["/pets"]["post"];

/** What `createPet` receives: POST /pets. */
export interface CreatePetInput {
  readonly body: NonNullable<CreatePetOperation["requestBody"]>["content"]["application/json"];
}

/** What `createPet` answers: one member per response the spec declares. */
export type CreatePetOutput =
  | { readonly status: 201; readonly body: CreatePetOperation["responses"][201]["content"]["application/json"]; readonly headers?: Readonly<Record<string, string>> }
  | { readonly status: 400; readonly body: CreatePetOperation["responses"][400]["content"]["application/problem+json"]; readonly headers?: Readonly<Record<string, string>> };

type GetPetsOperation = paths["/pets"]["get"];

/** What `getPets` receives: GET /pets. */
export interface GetPetsInput {
  readonly query: NonNullable<GetPetsOperation["parameters"]["query"]>;
}

/** What `getPets` answers: one member per response the spec declares. */
export type GetPetsOutput =
  | { readonly status: 200; readonly body: GetPetsOperation["responses"][200]["content"]["application/json"]; readonly headers?: Readonly<Record<string, string>> };

type GetPetOperation = paths["/pets/{petId}"]["get"];

/** What `getPet` receives: GET /pets/{petId}. */
export interface GetPetInput {
  readonly path: NonNullable<GetPetOperation["parameters"]["path"]>;
}

/** What `getPet` answers: one member per response the spec declares. */
export type GetPetOutput =
  | { readonly status: 200; readonly body: GetPetOperation["responses"][200]["content"]["application/json"]; readonly headers?: Readonly<Record<string, string>> }
  | { readonly status: 404; readonly body?: undefined; readonly headers?: Readonly<Record<string, string>> };

/**
 * Every `pets` route. Extend it with a class decorated `@Controller()` and implement each method:
 * a new operation in the spec is a new abstract method, so the extending class stops compiling until it has one.
 */
export abstract class PetsControllerBase {
  /** POST /pets. */
  abstract createPet(input: CreatePetInput): CreatePetOutput | Promise<CreatePetOutput>;

  /** GET /pets. */
  abstract getPets(input: GetPetsInput): GetPetsOutput | Promise<GetPetsOutput>;

  /** GET /pets/{petId}. */
  abstract getPet(input: GetPetInput): GetPetOutput | Promise<GetPetOutput>;

  /** POST /pets: validates the request, calls `createPet`, and checks what it answers. */
  @Post("/pets")
  async handleCreatePet(@Body() body: unknown, @Res({ passthrough: true }) reply: Reply): Promise<unknown> {
    const input = {
      body,
    };
    const errors = invalid([
      { pointer: "/body", value: input.body, required: true, check: validators["createPet"]["body:application/json"] },
    ]);
    if (errors.length > 0) return problem(reply, 400, 'Bad Request', 'The request does not match the contract.', errors);
    return send(reply, await this.createPet(input as unknown as CreatePetInput), {
      "201": { media: "application/json", check: validators["createPet"]["response:201:application/json"] },
      "400": { media: "application/problem+json", check: validators["createPet"]["response:400:application/problem+json"] },
    });
  }

  /** GET /pets: validates the request, calls `getPets`, and checks what it answers. */
  @Get("/pets")
  async handleGetPets(@Query() query: Record<string, unknown>, @Res({ passthrough: true }) reply: Reply): Promise<unknown> {
    const input = {
      query: { "limit": coerce(query["limit"], "number") },
    };
    const errors = invalid([
      { pointer: "/query/limit", value: input.query["limit"], required: false, check: validators["get /pets"]["param:query:limit"] },
    ]);
    if (errors.length > 0) return problem(reply, 400, 'Bad Request', 'The request does not match the contract.', errors);
    return send(reply, await this.getPets(input as unknown as GetPetsInput), {
      "200": { media: "application/json", check: validators["get /pets"]["response:200:application/json"] },
    });
  }

  /** GET /pets/{petId}: validates the request, calls `getPet`, and checks what it answers. */
  @Get("/pets/:petId")
  async handleGetPet(@Param() path: Record<string, unknown>, @Res({ passthrough: true }) reply: Reply): Promise<unknown> {
    const input = {
      path: { "petId": coerce(path["petId"], "number") },
    };
    const errors = invalid([
      { pointer: "/path/petId", value: input.path["petId"], required: true, check: validators["getPet"]["param:path:petId"] },
    ]);
    if (errors.length > 0) return problem(reply, 400, 'Bad Request', 'The request does not match the contract.', errors);
    return send(reply, await this.getPet(input as unknown as GetPetInput), {
      "200": { media: "application/json", check: validators["getPet"]["response:200:application/json"] },
      "404": {},
    });
  }
}
